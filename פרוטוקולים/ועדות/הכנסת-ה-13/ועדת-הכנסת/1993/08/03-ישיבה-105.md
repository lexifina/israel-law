בקשת היועץ המשפטי לממשלה לנטילת החסינות מהשר אריה דרעי
היו"ר חי מירום;

נוהל, בירור העניינים שלפנינו ועדיין לא לצורך כניסה לעומקה של הבקשה עצמה
ולעומקן של הגנות אלה או אחרות, מתוך כוונה ללמוד קודם מהיועץ המשפטי לממשלה
דברים שנדרשים בנסיבות הענין על שיקוליו בהגשת כתב האישום בשלב זה כמו: האם כתב
האישום הזה הוא מלא בנושאים האישיים, לצורך מה נמשכת הוקירה, מדוע זה מוגש לנו
עכשיו לעת פגרה כאשר משהו אולי לא הסתיים. אלה שאלות שהעלו בפני חברי ועדה במהלך
התייעצויות לא פורמליות שקיימנו אתמול לצורך הבנת ההליך.
לארור שנשמע את היועץ המשפטי לממשלה, ואולי גם את פרקליטת המדינה שתרצה להוסיף
דברים בקטע הזה - עדיין לא טיעון לגופה של הבקשה להסרת החסינות - של השאלות
הנילוות להגשת כתב האישום, אאפשר לשר אריה דרעי ולחברי הכנסת להעלות שאלות ליועץ
המשפטי לממשלה. אחר-כך כולנו, היועץ המשפטי לממשלה, השר דרעי ואנחנו ננסה לקבוע
את הנוהל ולוח-הזמנים שלפיו תעבוד הוועדה בעיקר בנסיבות המיוחדות של הפגרה
שלפנינו.

של היום תיסוב בעיקר סביב נוהל עבודה של הוועדה, קביעת לוח זמנים ודרכי עבודתה
בניהול הנושא העומד בפני נו.
ועדת הכנסת עיצבה נוהל טוב, תקין ומסודר במשך השנים. ריעננו אותו כאשר
נכנסנו לדיון בענין הסרת החסינות של חבר-הכנסת פנחסי וקבענו אז כמה כללים שהינחו
אותנו, עבדנו על-פיהם ויצרנו הליך נכון ותקין בוועדת הכנסת. עובדה היא שחרף
הנסיון לתקוף את התהליכים בוועדת הכנסת, בג"צ לא מצא לנחוץ להתערב כהוא זה בתהליך
שניהלנו בוועדה והיו גם אמרות-אגב בזכות התהליך הזה. להבדיל מתהליכים אחרים שהיו
במליאה, התהליך בוועדה היה נכון פחות או יותר. החלטנו אז לקבוע לוח זמנים וזה
היה יותר קל מאשר היום כי עכשיו אנחנו עומדים לפיתחה של פגרה ויש כאן ענייניים
בעייתיים בקביעת לוח זמנים. קבענו אז את אופן השתתפותם של חברי הוועדה בדיונים
של הסרת חסינות. הכלל היה שאלה שמתחילים חם גם אלה שמסיימים ואם יש הבר כנסת
שיודע שבתקופה נתונה בה אנחנו מקיימים דיון הוא איננו יכול להשתתף בישיבות, שומה
עליו להודיע על כך מראש כדי לאפשר לסיעתו להחליף אותו על-מנת שמי שיושב בהליך
מעין שיפוטי - על כך אין ויכוח - יוכל להימצא בכל ישיבה מן הישיבות.

ר' מילוא;
יש אומדו של הזמנים?
היו"ר חי מירום;
מנסיון העבר אנחנו יודעים שמסגרת הזמן לא צריכה להשתרע על יותר מאשר 6-5
ישיבות ואנחנו משתדלים להקפיד על כך שמידת הזמן שתינתן לכל צד - ליועץ המשפטי
לממשלה מצד אחד, ולחבר הכנסת שמבקשים להסיר את חסינותו ולבא כוחו מצד שני - תהיה
פחות או יותר שווה על-מנת שהצדק לא רק ייעשה אלא גם ייראה, ובעבר עמדנו בכך.
החלטנו אז שאנחנו רוצים לקבוע ישיבות שוטפות, רצופות וארוכות מן הרגיל, היו
ישיבות שארכו 6 שעות לישיבה, ובלבד שלא נמשוך את התהליך יתר על המידה ולא נהפוך
אותו לסחבת כי לחברי הכנסת קשה לעמוד בזה מבחינת ההתחייבויות שלהם. הקפדנו שלא
תהיינה התחלפויות וקבענו שלא נקבל אפילו מחשבה על משמעת סיעתית או משהו מעין זה
וכי שיקול-הדעת של החברים, ללא הבדל סיעות, הוא שיקול-דעת אישי, לא נסבול אווירת
לחצים ולא לחצים שיש בהם שיקול פוליטי או אחר. אמרנו את הדברים האלה לעצמנו
בבחינת הוראה ועמדנו בזה יפה, ההצבעות הוכיחו שכל אחד כאן פעל לפי מצפונו.
את הכללים האלה אנחנו צריכים לבסס גם כאן אבל אנחנו יוצאים לפגרה ויש בעיות
שקשורות ליכולתו של השר דרעי להגן על עצמו. הוא מופיע באן היום ללא בא-כוח ועוד
מעט הוא יסביר מת הן בעיות הייצוג מבחינתו. אינני אומר שהכנסת חייבת להתאים את
עצמה ללוח זמנים של כל פרקליט, אבל אין ספק שהכנסת צריכה להתחשב בשאלת הייצוג של
חבר מחבריה וביכולתו להתגונן.

יכולים לשאוב ממנו את הידיעה הוודאית שמדובר כאן בענין הקונקרטי. אם מדובר בענין
הקונקרטי זה אומר שעל-פי מצוות בית-המשפט אנחנו הולכים לחוקק חוק שבא להתערב
בהליך שתלוי ועומד בבית-המשפט ולשנות זכויות של אנשים - התוצאה היא שבית-המשפט
מחכה להחלטת הכנסת.
אני הבנתי, כך היה כתוב בדברי ההסבר, שבית-המשפט כבר פסק לענין לגופו וכי
האוצר מבקש שהדברים יעוגנו בחוק. כפי שציין חבר-הכנסת דמביץ - כך אנחנו קוראים
לו בוועדת הכספים בגלל תרומתו לבית - הענין הוא מהותי וצריך לתת עליו את הדעת.
אני רוצה שנישאר עם כל האופציות ביד; אם יש שיקול של דחיפות - עדיין נופל לשחרר
מחובת הנחה; אם לא - שיתנו לנו את הזמן לשבת על החוק.
היו"ר חי מירום;
חבר-הכנסת איתן, אני מקבל את טיעונך ולכן נקדיש לכך ישיבה שלמה היום
אחר-הצהריים.
מי איתן;
אי-אפשר לקיים אותה מחר?

היו"ר ח' מירום;
אם זה יהיה מחר זה מסכל את הכוונה.
סגנית מזכיר הכנסת ש' כרם;
זו קריאה שניה וקריאה שלישית.
אי פורז;
גם אני חבר ועדת הכספים ולא אתן יד לחקיקה חפוזה. זו רק אופציה.
היו"ר חי מירום:
העניו הובהר. נזמין את היועץ המשפטי של ועדת הכספים לישיבה מחר בבוקר. אם
ירצה לבוא וישכנע אותנו - נפטור מחובת הנחה; אם לא ירצה לבוא - לא נפטור מחובת
הנחה; אם ירצה, יבוא ולא ישכנע - לא נפטור מחובת הנחה.
אני מודה לכולם.
